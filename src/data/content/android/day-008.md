```java
package com.example.smartmeter.db;

import android.content.Context;
import androidx.room.Database;
import androidx.room.Room;
import androidx.room.RoomDatabase;
import com.example.smartmeter.model.MeterReadingEntity;

@Database(entities = {MeterReadingEntity.class}, version = 1)
public abstract class AppDatabase extends RoomDatabase {
    public abstract MeterReadingDao meterReadingDao();

    private static volatile AppDatabase INSTANCE;

    public static AppDatabase getInstance(Context context) {
        if (INSTANCE == null) {
            synchronized (AppDatabase.class) {
                if (INSTANCE == null) {
                    INSTANCE = Room.databaseBuilder(context.getApplicationContext(),
                                    AppDatabase.class, "meter_database")
                            .build();
                }
            }
        }
        return INSTANCE;
    }
}
```

**`TokenManager.java`** (Day 5 — wraps EncryptedSharedPreferences):
```java
package com.example.smartmeter.util;

import android.content.Context;
import android.content.SharedPreferences;
import androidx.security.crypto.EncryptedSharedPreferences;
import androidx.security.crypto.MasterKey;
import java.security.GeneralSecurityException;
import java.io.IOException;

public class TokenManager {
    private final SharedPreferences prefs;

    public TokenManager(Context context) {
        try {
            MasterKey masterKey = new MasterKey.Builder(context)
                    .setKeyScheme(MasterKey.KeyScheme.AES256_GCM)
                    .build();
            prefs = EncryptedSharedPreferences.create(
                    context, "secure_prefs", masterKey,
                    EncryptedSharedPreferences.PrefKeyEncryptionScheme.AES256_SIV,
                    EncryptedSharedPreferences.PrefValueEncryptionScheme.AES256_GCM
            );
        } catch (GeneralSecurityException | IOException e) {
            throw new RuntimeException("Secure storage init failed", e);
        }
    }

    public void saveToken(String token, long expiresAt) {
        prefs.edit().putString("jwt_token", token).putLong("expires_at", expiresAt).apply();
    }

    public String getToken() {
        return prefs.getString("jwt_token", null);
    }

    public boolean isLoggedIn() {
        String token = getToken();
        long expiresAt = prefs.getLong("expires_at", 0);
        return token != null && System.currentTimeMillis() < expiresAt;
    }

    public void clear() {
        prefs.edit().clear().apply();
    }
}
```

**`NetworkUtils.java`** (Day 6):
```java
package com.example.smartmeter.util;

import android.content.Context;
import android.net.ConnectivityManager;
import android.net.Network;
import android.net.NetworkCapabilities;

public class NetworkUtils {
    public static boolean isOnline(Context context) {
        ConnectivityManager cm = (ConnectivityManager)
                context.getSystemService(Context.CONNECTIVITY_SERVICE);
        Network network = cm.getActiveNetwork();
        if (network == null) return false;
        NetworkCapabilities capabilities = cm.getNetworkCapabilities(network);
        return capabilities != null &&
               (capabilities.hasTransport(NetworkCapabilities.TRANSPORT_WIFI) ||
                capabilities.hasTransport(NetworkCapabilities.TRANSPORT_CELLULAR));
    }
}
```

## Step 5: API Layer — `api/`

**`AuthInterceptor.java`** (Day 4+5 — auto-attaches JWT):
```java
package com.example.smartmeter.api;

import androidx.annotation.NonNull;
import com.example.smartmeter.util.TokenManager;
import java.io.IOException;
import okhttp3.Interceptor;
import okhttp3.Request;
import okhttp3.Response;

public class AuthInterceptor implements Interceptor {
    private final TokenManager tokenManager;

    public AuthInterceptor(TokenManager tokenManager) {
        this.tokenManager = tokenManager;
    }

    @NonNull
    @Override
    public Response intercept(@NonNull Chain chain) throws IOException {
        Request original = chain.request();
        String token = tokenManager.getToken();

        if (token == null) return chain.proceed(original);

        Request authorized = original.newBuilder()
                .header("Authorization", "Bearer " + token)
                .build();
        return chain.proceed(authorized);
    }
}
```

**`RetrofitClient.java`** (Day 4 — OkHttp + Interceptor wired together):
```java
package com.example.smartmeter.api;

import android.content.Context;
import com.example.smartmeter.util.TokenManager;
import okhttp3.OkHttpClient;
import okhttp3.logging.HttpLoggingInterceptor;
import retrofit2.Retrofit;
import retrofit2.converter.gson.GsonConverterFactory;
import java.util.concurrent.TimeUnit;

public class RetrofitClient {
    private static Retrofit retrofit;
    private static final String BASE_URL = "https://api.smartmeter.com/";

    public static Retrofit getInstance(Context context) {
        if (retrofit == null) {
            HttpLoggingInterceptor logging = new HttpLoggingInterceptor();
            logging.setLevel(HttpLoggingInterceptor.Level.BODY);

            OkHttpClient client = new OkHttpClient.Builder()
                    .addInterceptor(new AuthInterceptor(new TokenManager(context.getApplicationContext())))
                    .addInterceptor(logging)
                    .connectTimeout(30, TimeUnit.SECONDS)
                    .readTimeout(30, TimeUnit.SECONDS)
                    .build();

            retrofit = new Retrofit.Builder()
                    .baseUrl(BASE_URL)
                    .client(client)
                    .addConverterFactory(GsonConverterFactory.create())
                    .build();
        }
        return retrofit;
    }
}
```

## Step 6: Repository Layer — `repository/`

**`AuthRepository.java`** (Day 5):
```java
package com.example.smartmeter.repository;

import android.content.Context;
import com.example.smartmeter.api.*;
import com.example.smartmeter.model.*;
import com.example.smartmeter.util.TokenManager;
import retrofit2.Call;
import retrofit2.Callback;
import retrofit2.Response;

public class AuthRepository {
    private final AuthApiService api;
    private final TokenManager tokenManager;

    public AuthRepository(Context context) {
        api = RetrofitClient.getInstance(context).create(AuthApiService.class);
        tokenManager = new TokenManager(context.getApplicationContext());
    }

    public void login(String email, String password, AuthCallback callback) {
        api.login(new LoginRequest(email, password)).enqueue(new Callback<LoginResponse>() {
            @Override
            public void onResponse(Call<LoginResponse> call, Response<LoginResponse> response) {
                if (response.isSuccessful() && response.body() != null) {
                    tokenManager.saveToken(response.body().token, response.body().expiresAt);
                    callback.onSuccess();
                } else {
                    callback.onError("Invalid credentials");
                }
            }
            @Override
            public void onFailure(Call<LoginResponse> call, Throwable t) {
                callback.onError("Network error: " + t.getMessage());
            }
        });
    }

    public boolean isLoggedIn() {
        return tokenManager.isLoggedIn();
    }

    public void logout() {
        tokenManager.clear();
    }

    public interface AuthCallback {
        void onSuccess();
        void onError(String message);
    }
}
```

**`MeterRepository.java`** (Day 6 — the offline-first core):
```java
package com.example.smartmeter.repository;

import android.content.Context;
import androidx.lifecycle.LiveData;
import com.example.smartmeter.api.*;
import com.example.smartmeter.db.*;
import com.example.smartmeter.model.*;
import com.example.smartmeter.util.NetworkUtils;
import com.example.smartmeter.util.SyncStatus;
import retrofit2.Call;
import retrofit2.Callback;
import retrofit2.Response;
import java.util.UUID;
import java.util.concurrent.ExecutorService;
import java.util.concurrent.Executors;

public class MeterRepository {
    private final MeterApiService api;
    private final MeterReadingDao dao;
    private final Context appContext;
    private final ExecutorService executor = Executors.newSingleThreadExecutor();

    public MeterRepository(Context context) {
        appContext = context.getApplicationContext();
        api = RetrofitClient.getInstance(context).create(MeterApiService.class);
        dao = AppDatabase.getInstance(context).meterReadingDao();
    }

    public void saveReading(String meterId, double reading) {
        MeterReadingEntity entity = new MeterReadingEntity();
        entity.clientUuid = UUID.randomUUID().toString();
        entity.meterId = meterId;
        entity.reading = reading;
        entity.timestamp = System.currentTimeMillis();
        entity.syncStatus = SyncStatus.PENDING;

        executor.execute(() -> {
            dao.insert(entity);
            if (NetworkUtils.isOnline(appContext)) {
                attemptSync(entity);
            }
        });
    }

    private void attemptSync(MeterReadingEntity entity) {
        MeterReadingRequest request = new MeterReadingRequest(
                entity.clientUuid, entity.meterId, entity.reading, entity.timestamp);

        api.submitReading(request).enqueue(new Callback<MeterReadingResponse>() {
            @Override
            public void onResponse(Call<MeterReadingResponse> call, Response<MeterReadingResponse> response) {
                entity.syncStatus = response.isSuccessful() ? SyncStatus.SYNCED : SyncStatus.FAILED;
                executor.execute(() -> dao.update(entity));
            }
            @Override
            public void onFailure(Call<MeterReadingResponse> call, Throwable t) {
                entity.syncStatus = SyncStatus.FAILED;
                executor.execute(() -> dao.update(entity));
            }
        });
    }

    public java.util.List<MeterReadingEntity> getReadingsForMeter(String meterId) {
        return dao.getReadingsForMeter(meterId);
    }

    public void syncAllPending() {
        executor.execute(() -> {
            for (MeterReadingEntity e : dao.getReadingsByStatus(SyncStatus.PENDING)) attemptSync(e);
            for (MeterReadingEntity e : dao.getReadingsByStatus(SyncStatus.FAILED)) attemptSync(e);
        });
    }
}
```

> **Yaad Rakho:** `getReadingsForMeter` yahan seedha list return karta hai (background thread ke saath call karna hoga). Chahe to Day 7 ke `LiveData` pattern extend karke Room ka `LiveData<List<...>>` return type bhi use kar sakte ho — ViewModel me observe karna aur bhi seedha ho jaata hai.

```java
package com.example.smartmeter.ui;

import android.content.Intent;
import android.os.Bundle;
import androidx.appcompat.app.AppCompatActivity;
import com.example.smartmeter.databinding.ActivityLoginBinding;
import com.example.smartmeter.repository.AuthRepository;

public class LoginActivity extends AppCompatActivity {
    private ActivityLoginBinding binding;
    private AuthRepository authRepository;

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        authRepository = new AuthRepository(this);

        if (authRepository.isLoggedIn()) {
            startActivity(new Intent(this, DashboardActivity.class));
            finish();
            return;
        }

        binding = ActivityLoginBinding.inflate(getLayoutInflater());
        setContentView(binding.getRoot());

        binding.btnLogin.setOnClickListener(v -> {
            String email = binding.etEmail.getText().toString();
            String password = binding.etPassword.getText().toString();

            authRepository.login(email, password, new AuthRepository.AuthCallback() {
                @Override
                public void onSuccess() {
                    startActivity(new Intent(LoginActivity.this, DashboardActivity.class));
                    finish();
                }
                @Override
                public void onError(String message) {
                    binding.etPassword.setError(message);
                }
            });
        });
    }
}
```

```java
package com.example.smartmeter.ui;

import android.os.Bundle;
import androidx.appcompat.app.AppCompatActivity;
import androidx.lifecycle.ViewModelProvider;
import androidx.recyclerview.widget.LinearLayoutManager;
import androidx.work.*;
import com.example.smartmeter.databinding.ActivityDashboardBinding;
import com.example.smartmeter.repository.MeterRepository;
import com.example.smartmeter.viewmodel.*;
import com.example.smartmeter.worker.SyncWorker;
import java.util.concurrent.TimeUnit;

public class DashboardActivity extends AppCompatActivity {
    private ActivityDashboardBinding binding;
    private MeterViewModel viewModel;
    private static final String METER_ID = "MTR-001";

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        binding = ActivityDashboardBinding.inflate(getLayoutInflater());
        setContentView(binding.getRoot());

        MeterRepository repository = new MeterRepository(getApplicationContext());
        viewModel = new ViewModelProvider(this, new MeterViewModelFactory(repository))
                .get(MeterViewModel.class);

        MeterReadingAdapter adapter = new MeterReadingAdapter(viewModel.getReadings(METER_ID));
        binding.recyclerView.setLayoutManager(new LinearLayoutManager(this));
        binding.recyclerView.setAdapter(adapter);

        binding.btnSaveReading.setOnClickListener(v -> {
            String text = binding.etReading.getText().toString();
            if (text.isEmpty()) return;
            viewModel.saveReading(METER_ID, Double.parseDouble(text));
            binding.etReading.setText("");
            adapter.notifyDataSetChanged(); // simple refresh — Day 6 ke DiffUtil se optimize kar sakte ho
        });

        schedulePeriodicSync();
    }

    private void schedulePeriodicSync() {
        Constraints constraints = new Constraints.Builder()
                .setRequiredNetworkType(NetworkType.CONNECTED)
                .build();
        PeriodicWorkRequest syncRequest = new PeriodicWorkRequest.Builder(SyncWorker.class, 15, TimeUnit.MINUTES)
                .setConstraints(constraints)
                .build();
        WorkManager.getInstance(this)
                .enqueueUniquePeriodicWork("meter_sync", ExistingPeriodicWorkPolicy.KEEP, syncRequest);
    }
}
```

## Step 10: Pehli Baar Run Karo

1. **Gradle sync** clean hona chahiye (koi red error nahi Build panel me).
2. **Run ▶️** dabao — `LoginActivity` khulni chahiye (kyunki wo Manifest me `MAIN`/`LAUNCHER` hai).
3. Login karo (server na bhi ho, `onError` message "Invalid credentials" ya "Network error" dikhega — ye batata hai error handling kaam kar raha hai, chahe real backend na ho).
4. Agar apna test backend nahi hai, `RetrofitClient`'s `BASE_URL` ko [jsonplaceholder.typicode.com](https://jsonplaceholder.typicode.com) jaisi kisi test API se badal ke pehle **network layer** independently test karo (Day 4 ka Hands-On Lab yaad hai).

### Common Assembly Errors

| Error | Fix |
|-------|-----|
| `Cannot resolve symbol 'ActivityLoginBinding'` | View Binding `build.gradle` me `true` set hai ya nahi check karo, phir **Build → Rebuild Project** |
| `Room cannot verify the data integrity` | App uninstall karke dobara install karo (schema version mismatch — dev me normal hai) |
| `NetworkOnMainThreadException` | Kahin `.execute()` (Retrofit sync call) ya Room query bina `executor.execute()` ke main thread pe chal rahi hai |
| `IllegalStateException: WorkManager is not initialized` | `androidx.work:work-runtime` dependency check karo, Gradle sync dobara karo |
| App crash on `AppCompatActivity` not found | `androidx.appcompat:appcompat` dependency missing — Step 0 dubara check karo |
