```dart
// lib/features/notes/domain/entities/note.dart
import 'package:equatable/equatable.dart';

class Note extends Equatable {
  final String id;
  final String title;
  final String content;
  final List<String> tags;
  final DateTime createdAt;
  final DateTime updatedAt;
  final String userId;

  const Note({
    required this.id,
    required this.title,
    required this.content,
    required this.tags,
    required this.createdAt,
    required this.updatedAt,
    required this.userId,
  });

  Note copyWith({
    String? title,
    String? content,
    List<String>? tags,
    DateTime? updatedAt,
  }) {
    return Note(
      id: id,
      title: title ?? this.title,
      content: content ?? this.content,
      tags: tags ?? this.tags,
      createdAt: createdAt,
      updatedAt: updatedAt ?? this.updatedAt,
      userId: userId,
    );
  }

  @override
  List<Object> get props => [id, title, content, tags, createdAt, updatedAt, userId];
}
```

## Step 4: Breakpoints + Adaptive Layout

```dart
// lib/core/utils/breakpoints.dart
import 'package:flutter/material.dart';

class Breakpoints {
  static const double mobile = 600;
  static const double tablet = 1024;

  static bool isMobile(BuildContext c) => MediaQuery.of(c).size.width < mobile;
  static bool isTablet(BuildContext c) {
    final w = MediaQuery.of(c).size.width;
    return w >= mobile && w < tablet;
  }
  static bool isDesktop(BuildContext c) => MediaQuery.of(c).size.width >= tablet;
}
```

```dart
// lib/shared/widgets/adaptive_layout.dart
import 'package:flutter/material.dart';

class AdaptiveLayout extends StatelessWidget {
  final Widget mobile;
  final Widget? tablet;
  final Widget desktop;

  const AdaptiveLayout({
    super.key,
    required this.mobile,
    this.tablet,
    required this.desktop,
  });

  @override
  Widget build(BuildContext context) {
    return LayoutBuilder(
      builder: (context, constraints) {
        if (constraints.maxWidth >= 1024) return desktop;
        if (constraints.maxWidth >= 600) return tablet ?? mobile;
        return mobile;
      },
    );
  }
}
```

## Step 5: Platform-Specific Code

```dart
// lib/core/utils/platform.dart
import 'package:flutter/foundation.dart';
import 'dart:io' show Platform;

class AppPlatform {
  static bool get isWeb => kIsWeb;
  static bool get isMobile => !kIsWeb && (Platform.isAndroid || Platform.isIOS);
  static bool get isDesktop => !kIsWeb && (Platform.isWindows || Platform.isMacOS || Platform.isLinux);

  static String get name {
    if (kIsWeb) return 'Web';
    if (Platform.isAndroid) return 'Android';
    if (Platform.isIOS) return 'iOS';
    if (Platform.isWindows) return 'Windows';
    if (Platform.isMacOS) return 'macOS';
    if (Platform.isLinux) return 'Linux';
    return 'Unknown';
  }
}
```

```dart
// lib/features/notes/presentation/screens/notes_list_screen.dart
import 'package:flutter/material.dart';
import '../../domain/entities/note.dart';
import '../../../../shared/widgets/adaptive_layout.dart';
import 'note_detail_screen.dart';

class NotesListScreen extends StatefulWidget {
  const NotesListScreen({super.key});

  @override
  State<NotesListScreen> createState() => _NotesListScreenState();
}

class _NotesListScreenState extends State<NotesListScreen> {
  final List<Note> _notes = [
    Note(
      id: '1',
      title: 'Flutter Notes',
      content: 'Learn Bloc + Clean Architecture',
      tags: const ['flutter'],
      createdAt: DateTime.now(),
      updatedAt: DateTime.now(),
      userId: 'demo',
    ),
    Note(
      id: '2',
      title: 'Project Ideas',
      content: 'Build a multi-platform notes app',
      tags: const ['ideas'],
      createdAt: DateTime.now(),
      updatedAt: DateTime.now(),
      userId: 'demo',
    ),
  ];

  Note? _selectedNote;

  @override
  Widget build(BuildContext context) {
    return AdaptiveLayout(
      mobile: _MobileView(notes: _notes),
      desktop: _DesktopView(
        notes: _notes,
        selectedNote: _selectedNote,
        onSelectNote: (note) => setState(() => _selectedNote = note),
      ),
    );
  }
}

class _MobileView extends StatelessWidget {
  final List<Note> notes;
  const _MobileView({required this.notes});

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('Notes')),
      body: ListView.builder(
        itemCount: notes.length,
        itemBuilder: (_, i) => Card(
          margin: const EdgeInsets.symmetric(horizontal: 12, vertical: 4),
          child: ListTile(
            title: Text(notes[i].title, style: const TextStyle(fontWeight: FontWeight.bold)),
            subtitle: Text(notes[i].content, maxLines: 2, overflow: TextOverflow.ellipsis),
            trailing: const Icon(Icons.chevron_right),
            onTap: () => Navigator.push(context, MaterialPageRoute(builder: (_) => NoteDetailScreen(note: notes[i]))),
          ),
        ),
      ),
      floatingActionButton: FloatingActionButton(onPressed: () {}, child: const Icon(Icons.add)),
    );
  }
}

class _DesktopView extends StatelessWidget {
  final List<Note> notes;
  final Note? selectedNote;
  final ValueChanged<Note> onSelectNote;

  const _DesktopView({required this.notes, required this.selectedNote, required this.onSelectNote});

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      body: Row(
        children: [
          // Sidebar
          Container(
            width: 320,
            color: Theme.of(context).colorScheme.surfaceVariant,
            child: Column(
              children: [
                Container(
                  padding: const EdgeInsets.all(16),
                  child: Row(
                    mainAxisAlignment: MainAxisAlignment.spaceBetween,
                    children: [
                      const Text('Notes', style: TextStyle(fontSize: 24, fontWeight: FontWeight.bold)),
                      IconButton(icon: const Icon(Icons.add), onPressed: () {}),
                    ],
                  ),
                ),
                Expanded(
                  child: ListView.builder(
                    itemCount: notes.length,
                    itemBuilder: (_, i) {
                      final note = notes[i];
                      final selected = selectedNote?.id == note.id;
                      return ListTile(
                        title: Text(note.title, style: TextStyle(fontWeight: selected ? FontWeight.bold : FontWeight.normal)),
                        subtitle: Text(note.content, maxLines: 1, overflow: TextOverflow.ellipsis),
                        selected: selected,
                        onTap: () => onSelectNote(note),
                      );
                    },
                  ),
                ),
              ],
            ),
          ),
          const VerticalDivider(width: 1),

          // Content area
          Expanded(
            child: selectedNote == null
                ? const Center(child: Text('Select a note to view'))
                : NoteDetailScreen(note: selectedNote!),
          ),
        ],
      ),
    );
  }
}
```
