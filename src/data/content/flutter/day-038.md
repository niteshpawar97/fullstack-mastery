# 🌅 HOUR 1 — CONCEPT

## Platform Channels

Flutter ke 80% APIs Dart me hain — but kuch native features (battery, system services, native libraries) Dart me nahi. **Method Channels** se Dart aur native (Kotlin/Swift) communicate karte hain.

> **Socho Aise:** Two ghar friends — alag languages bolte hain. **Translator** beech me — message convert karke deta hai. Method Channel = translator between Dart and native.

---

## Architecture

```
Flutter (Dart)
     ↓ MethodChannel.invokeMethod('getBattery')
[Platform Channel]
     ↓
Native (Android Kotlin / iOS Swift)
     ↓ register handler
Returns value back to Flutter
```

---

## Step 1: Dart Side

```dart
import 'package:flutter/services.dart';

class BatteryService {
  static const _channel = MethodChannel('com.example.app/battery');

  static Future<int> getBatteryLevel() async {
    try {
      final int level = await _channel.invokeMethod('getBatteryLevel');
      return level;
    } on PlatformException catch (e) {
      throw Exception('Failed: ${e.message}');
    }
  }
}

// Usage
final level = await BatteryService.getBatteryLevel();
print('Battery: $level%');
```

> **Channel name** unique hona chahiye — typically `com.<your-domain>.<app>/<feature>`.

## Step 2: Android (Kotlin)

`android/app/src/main/kotlin/com/example/app/MainActivity.kt`:

```kotlin
package com.example.app

import android.content.Context
import android.content.Intent
import android.content.IntentFilter
import android.os.BatteryManager
import android.os.Build
import io.flutter.embedding.android.FlutterActivity
import io.flutter.embedding.engine.FlutterEngine
import io.flutter.plugin.common.MethodChannel

class MainActivity: FlutterActivity() {
    private val CHANNEL = "com.example.app/battery"

    override fun configureFlutterEngine(flutterEngine: FlutterEngine) {
        super.configureFlutterEngine(flutterEngine)

        MethodChannel(flutterEngine.dartExecutor.binaryMessenger, CHANNEL)
            .setMethodCallHandler { call, result ->
                when (call.method) {
                    "getBatteryLevel" -> {
                        val level = getBatteryLevel()
                        if (level != -1) result.success(level)
                        else result.error("UNAVAILABLE", "Battery info unavailable", null)
                    }
                    "getDeviceModel" -> {
                        result.success(Build.MODEL)
                    }
                    else -> result.notImplemented()
                }
            }
    }

    private fun getBatteryLevel(): Int {
        val bm = getSystemService(Context.BATTERY_SERVICE) as BatteryManager
        return bm.getIntProperty(BatteryManager.BATTERY_PROPERTY_CAPACITY)
    }
}
```

## Step 3: iOS (Swift)

`ios/Runner/AppDelegate.swift`:

```swift
import Flutter
import UIKit

@UIApplicationMain
@objc class AppDelegate: FlutterAppDelegate {
    override func application(
        _ application: UIApplication,
        didFinishLaunchingWithOptions launchOptions: [UIApplication.LaunchOptionsKey: Any]?
    ) -> Bool {
        let controller = window?.rootViewController as! FlutterViewController
        let batteryChannel = FlutterMethodChannel(
            name: "com.example.app/battery",
            binaryMessenger: controller.binaryMessenger)

        batteryChannel.setMethodCallHandler { (call, result) in
            switch call.method {
            case "getBatteryLevel":
                self.receiveBatteryLevel(result: result)
            case "getDeviceModel":
                result(UIDevice.current.model)
            default:
                result(FlutterMethodNotImplemented)
            }
        }

        GeneratedPluginRegistrant.register(with: self)
        return super.application(application, didFinishLaunchingWithOptions: launchOptions)
    }

    private func receiveBatteryLevel(result: FlutterResult) {
        let device = UIDevice.current
        device.isBatteryMonitoringEnabled = true
        if device.batteryState == UIDevice.BatteryState.unknown {
            result(FlutterError(code: "UNAVAILABLE", message: "Battery info unavailable", details: nil))
        } else {
            result(Int(device.batteryLevel * 100))
        }
    }
}
```

## Pass Arguments

### Dart

```dart
final result = await _channel.invokeMethod('add', {'a': 5, 'b': 3});
```

### Kotlin

```kotlin
"add" -> {
    val a = call.argument<Int>("a") ?: 0
    val b = call.argument<Int>("b") ?: 0
    result.success(a + b)
}
```

### Swift

```swift
case "add":
    if let args = call.arguments as? [String: Int],
       let a = args["a"], let b = args["b"] {
        result(a + b)
    }
```

### Dart Side

```dart
// lib/services/native_service.dart
import 'package:flutter/services.dart';

class NativeService {
  static const _channel = MethodChannel('com.example.flutter_native/info');

  static Future<int> getBatteryLevel() async {
    try {
      return await _channel.invokeMethod<int>('getBatteryLevel') ?? 0;
    } on PlatformException catch (e) {
      throw Exception('Battery error: ${e.message}');
    }
  }

  static Future<String> getDeviceModel() async {
    try {
      return await _channel.invokeMethod<String>('getDeviceModel') ?? 'Unknown';
    } on PlatformException {
      return 'Unknown';
    }
  }

  static Future<bool> isCharging() async {
    try {
      return await _channel.invokeMethod<bool>('isCharging') ?? false;
    } on PlatformException {
      return false;
    }
  }

  static Future<Map<String, dynamic>> getDeviceInfo() async {
    try {
      final result = await _channel.invokeMethod<Map>('getDeviceInfo') ?? {};
      return Map<String, dynamic>.from(result);
    } on PlatformException {
      return {};
    }
  }
}
```

### MainActivity.kt (Full)

```kotlin
package com.example.flutter_native

import android.content.Context
import android.content.Intent
import android.content.IntentFilter
import android.os.BatteryManager
import android.os.Build
import io.flutter.embedding.android.FlutterActivity
import io.flutter.embedding.engine.FlutterEngine
import io.flutter.plugin.common.MethodChannel

class MainActivity: FlutterActivity() {
    private val CHANNEL = "com.example.flutter_native/info"

    override fun configureFlutterEngine(flutterEngine: FlutterEngine) {
        super.configureFlutterEngine(flutterEngine)

        MethodChannel(flutterEngine.dartExecutor.binaryMessenger, CHANNEL)
            .setMethodCallHandler { call, result ->
                when (call.method) {
                    "getBatteryLevel" -> {
                        val bm = getSystemService(Context.BATTERY_SERVICE) as BatteryManager
                        val level = bm.getIntProperty(BatteryManager.BATTERY_PROPERTY_CAPACITY)
                        result.success(level)
                    }
                    "getDeviceModel" -> result.success("${Build.MANUFACTURER} ${Build.MODEL}")
                    "isCharging" -> {
                        val intent = registerReceiver(null, IntentFilter(Intent.ACTION_BATTERY_CHANGED))
                        val status = intent?.getIntExtra(BatteryManager.EXTRA_STATUS, -1) ?: -1
                        result.success(status == BatteryManager.BATTERY_STATUS_CHARGING || status == BatteryManager.BATTERY_STATUS_FULL)
                    }
                    "getDeviceInfo" -> {
                        result.success(mapOf(
                            "model" to Build.MODEL,
                            "manufacturer" to Build.MANUFACTURER,
                            "androidVersion" to Build.VERSION.RELEASE,
                            "sdkInt" to Build.VERSION.SDK_INT,
                            "device" to Build.DEVICE,
                            "brand" to Build.BRAND
                        ))
                    }
                    else -> result.notImplemented()
                }
            }
    }
}
```

```swift
import Flutter
import UIKit

@UIApplicationMain
@objc class AppDelegate: FlutterAppDelegate {
    override func application(
        _ application: UIApplication,
        didFinishLaunchingWithOptions launchOptions: [UIApplication.LaunchOptionsKey: Any]?

                    result(Int(device.batteryLevel * 100))
                }
            case "getDeviceModel":
                result(UIDevice.current.model)
            case "isCharging":
                result(UIDevice.current.batteryState == .charging || UIDevice.current.batteryState == .full)
            case "getDeviceInfo":
                result([
                    "model": UIDevice.current.model,
                    "name": UIDevice.current.name,
                    "systemName": UIDevice.current.systemName,
                    "systemVersion": UIDevice.current.systemVersion
                ])
            default:
                result(FlutterMethodNotImplemented)
            }
        }

        return super.application(application, didFinishLaunchingWithOptions: launchOptions)
```

```dart
class NativeInfoScreen extends StatefulWidget {
  const NativeInfoScreen({super.key});
  @override
  State<NativeInfoScreen> createState() => _NativeInfoScreenState();
}

class _NativeInfoScreenState extends State<NativeInfoScreen> {
  int? _battery;
  bool? _charging;
  String? _model;
  Map<String, dynamic>? _info;

  Future<void> _load() async {
    final battery = await NativeService.getBatteryLevel();
    final model = await NativeService.getDeviceModel();
    final charging = await NativeService.isCharging();
    final info = await NativeService.getDeviceInfo();

    setState(() {
      _battery = battery;
      _model = model;
      _charging = charging;
      _info = info;
    });
  }

  @override
  void initState() {
    super.initState();
    _load();
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: const Text('Native Info'),
        actions: [IconButton(icon: const Icon(Icons.refresh), onPressed: _load)],

        padding: const EdgeInsets.all(16),
        children: [
```

## 📋 Quick Revision

| Step | Code |
|------|------|
| Dart channel | `MethodChannel('unique_id')` |
| Invoke | `_channel.invokeMethod<T>('method', args)` |
| Kotlin handler | `MethodChannel(...).setMethodCallHandler { call, result -> ... }` |
| Swift handler | `FlutterMethodChannel(...).setMethodCallHandler { call, result in ... }` |
| Pass args | `Map` from Dart, `call.argument<T>("key")` in native |
| Errors | `result.error(code, msg, details)` (native) → `PlatformException` (Dart) |
| Not impl | `result.notImplemented()` |

## ❌ Common Mistakes

```dart
// MISTAKE 1: Different channel names
Dart: MethodChannel('app/battery')
Kotlin: MethodChannel('app/Battery')  // ❌ case-sensitive

// MISTAKE 2: Wrong cast on native side
val a = call.argument<String>("a")  // arg is Int
// FIX: match types

// MISTAKE 3: PlatformException unhandled
await _channel.invoke...  // crashes on error
// FIX: try/catch on PlatformException

// MISTAKE 4: Heavy work blocking main thread
// Native side me long task → app freeze
// FIX: Coroutines (Kotlin) / DispatchQueue (Swift) async
```

## 🎯 Interview Tips

1. **Q: Method Channel vs platform-specific package?**
   *A: Channel — custom needs (proprietary SDK, native library). Package — common features (camera, location). Always check pub.dev first.*

2. **Q: Type conversion handle kaise hota hai?**
   *A: StandardMethodCodec — auto convert primitives, lists, maps. Custom types ke liye serialize JSON.*

3. **Q: Performance impact?**
   *A: Each invocation has overhead — frequent calls (60fps) avoid karo. Batch if possible.*
