## Step 1: services/orders_service.dart

```dart
import 'package:cloud_firestore/cloud_firestore.dart';
import 'package:firebase_auth/firebase_auth.dart';
import '../models/order.dart';

class OrdersService {
  final _firestore = FirebaseFirestore.instance;
  String get _userId => FirebaseAuth.instance.currentUser!.uid;
  CollectionReference get _ordersRef => _firestore.collection('orders');

  Future<String> placeOrder({
    required String restaurantId,
    required String restaurantName,
    required List<CartItem> items,
    required double total,
    required String address,
    required String phone,
  }) async {
    final docRef = await _ordersRef.add({
      'userId': _userId,
      'restaurantId': restaurantId,
      'restaurantName': restaurantName,
      'items': items.map((c) => c.toMap()).toList(),
      'total': total,
      'address': address,
      'phone': phone,
      'status': OrderStatus.pending.name,
      'createdAt': FieldValue.serverTimestamp(),
    });
    return docRef.id;
  }

  Stream<List<FoodOrder>> watchUserOrders() {
    return _ordersRef
        .where('userId', isEqualTo: _userId)
        .orderBy('createdAt', descending: true)
        .snapshots()
        .map((s) => s.docs.map((d) => FoodOrder.fromFirestore(d)).toList());
  }

  Stream<FoodOrder?> watchOrder(String orderId) {
    return _ordersRef.doc(orderId).snapshots().map((doc) {
      if (!doc.exists) return null;
      return FoodOrder.fromFirestore(doc);
    });
  }

  Future<void> cancelOrder(String orderId) async {
    await _ordersRef.doc(orderId).update({
      'status': OrderStatus.cancelled.name,
    });
  }
}

final ordersService = OrdersService();
```

## Step 3: services/fcm_service.dart

```dart
import 'package:cloud_firestore/cloud_firestore.dart';
import 'package:firebase_auth/firebase_auth.dart';
import 'package:firebase_messaging/firebase_messaging.dart';
import 'package:flutter_local_notifications/flutter_local_notifications.dart';

final localNotifications = FlutterLocalNotificationsPlugin();

class FcmService {
  final _messaging = FirebaseMessaging.instance;
  final _firestore = FirebaseFirestore.instance;

  Future<void> initialize() async {
    await _messaging.requestPermission(alert: true, badge: true, sound: true);

    // Init local notifications
    await localNotifications.initialize(
      const InitializationSettings(
        android: AndroidInitializationSettings('@mipmap/ic_launcher'),
        iOS: DarwinInitializationSettings(),
      ),
    );

    // Save token to Firestore
    final token = await _messaging.getToken();
    if (token != null) await _saveToken(token);
    _messaging.onTokenRefresh.listen(_saveToken);

    // Foreground listener — show as local notification
    FirebaseMessaging.onMessage.listen((message) {
      final title = message.notification?.title ?? 'Order Update';
      final body = message.notification?.body ?? '';

      localNotifications.show(
        DateTime.now().millisecondsSinceEpoch ~/ 1000,
        title,
        body,
        const NotificationDetails(
          android: AndroidNotificationDetails('orders_channel', 'Order Updates', importance: Importance.max),
          iOS: DarwinNotificationDetails(),
        ),
      );
    });
  }

  Future<void> _saveToken(String token) async {
    final uid = FirebaseAuth.instance.currentUser?.uid;
    if (uid == null) return;
    await _firestore.collection('users').doc(uid).set({
      'fcmToken': token,
    }, SetOptions(merge: true));
  }
}

final fcmService = FcmService();
```

## Step 4: Update cart_screen.dart Place Order

```dart
class _CartScreenState extends ConsumerState<CartScreen> {
  final _addressController = TextEditingController();
  final _phoneController = TextEditingController();
  bool _placing = false;

  Future<void> _placeOrder() async {
    if (_addressController.text.trim().isEmpty || _phoneController.text.trim().length != 10) {
      ScaffoldMessenger.of(context).showSnackBar(const SnackBar(content: Text('Address aur 10-digit phone do')));
      return;
    }

    final cart = ref.read(cartProvider);
    final total = ref.read(cartTotalProvider);
    final cartNotifier = ref.read(cartProvider.notifier);

    setState(() => _placing = true);

    try {
      final orderId = await ordersService.placeOrder(
        restaurantId: cartNotifier.restaurantId!,
        restaurantName: cartNotifier.restaurantName!,
        items: cart,
        total: total,
        address: _addressController.text.trim(),
        phone: _phoneController.text.trim(),
      );

      cartNotifier.clear();
      if (!mounted) return;
      Navigator.pushReplacement(
        context,
        MaterialPageRoute(builder: (_) => OrderTrackingScreen(orderId: orderId)),
      );
    } catch (e) {
      if (!mounted) return;
      ScaffoldMessenger.of(context).showSnackBar(SnackBar(content: Text('Order failed: $e')));
    } finally {
      if (mounted) setState(() => _placing = false);
    }
  }

  // Update place order button
  ElevatedButton(
    onPressed: _placing ? null : _placeOrder,
    child: _placing
      ? const CircularProgressIndicator(color: Colors.white)
      : const Text('PLACE ORDER (COD)'),
  )
}
```

```dart
import 'package:flutter/material.dart';
import 'package:flutter_riverpod/flutter_riverpod.dart';
import '../models/order.dart';

import '../services/orders_service.dart';

class OrderTrackingScreen extends ConsumerWidget {
  final String orderId;
  const OrderTrackingScreen({super.key, required this.orderId});

  @override
  Widget build(BuildContext context, WidgetRef ref) {
    final orderAsync = ref.watch(singleOrderProvider(orderId));

    return Scaffold(
      appBar: AppBar(title: const Text('Track Order')),
      body: orderAsync.when(
        data: (order) {
          if (order == null) return const Center(child: Text('Order nahi mila'));

          return SingleChildScrollView(
            padding: const EdgeInsets.all(16),
            child: Column(
              crossAxisAlignment: CrossAxisAlignment.stretch,
              children: [
                Container(
                  padding: const EdgeInsets.all(16),
                  decoration: BoxDecoration(color: Colors.indigo[50], borderRadius: BorderRadius.circular(12)),
                  child: Column(
                    crossAxisAlignment: CrossAxisAlignment.start,
                    children: [
                      Text('Order #${order.id.substring(0, 6).toUpperCase()}', style: const TextStyle(fontSize: 16, fontWeight: FontWeight.bold)),
                      Text(order.restaurantName),
                      Text('₹${order.total.toStringAsFixed(0)}', style: const TextStyle(fontSize: 20, fontWeight: FontWeight.bold)),
                    ],
                  ),
                ),
                const SizedBox(height: 24),
                _StatusTimeline(currentStatus: order.status),
                const SizedBox(height: 24),

                const Text('Items', style: TextStyle(fontSize: 18, fontWeight: FontWeight.bold)),
                const SizedBox(height: 8),
                ...order.items.map((c) => Padding(
                  padding: const EdgeInsets.symmetric(vertical: 4),
                  child: Row(children: [
                    Text('${c.quantity}× ', style: const TextStyle(fontWeight: FontWeight.bold)),
                    Expanded(child: Text(c.item.name)),
                    Text('₹${c.total.toStringAsFixed(0)}'),
                  ]),
                )),

                const Divider(height: 32),
                const Text('Delivery to', style: TextStyle(fontSize: 18, fontWeight: FontWeight.bold)),
                Text(order.address),
                Text('+91 ${order.phone}'),

                const SizedBox(height: 32),
                if (order.status == OrderStatus.pending)
                  OutlinedButton(
                    onPressed: () async {
                      await ordersService.cancelOrder(order.id);
                      if (context.mounted) {
                        ScaffoldMessenger.of(context).showSnackBar(const SnackBar(content: Text('Order cancelled')));
                      }
                    },
                    style: OutlinedButton.styleFrom(foregroundColor: Colors.red),
                    child: const Text('Cancel Order'),
                  ),
              ],
            ),
          );
        },
        loading: () => const Center(child: CircularProgressIndicator()),
        error: (e, _) => Center(child: Text('Error: $e')),
      ),
    );
  }
}

class _StatusTimeline extends StatelessWidget {
  final OrderStatus currentStatus;
  const _StatusTimeline({required this.currentStatus});

  @override
  Widget build(BuildContext context) {
    final steps = [
      (OrderStatus.pending, 'Order Placed', Icons.receipt),
      (OrderStatus.confirmed, 'Confirmed', Icons.check_circle),
      (OrderStatus.preparing, 'Preparing', Icons.restaurant),
      (OrderStatus.outForDelivery, 'On the way', Icons.delivery_dining),
      (OrderStatus.delivered, 'Delivered', Icons.home),
    ];

    final currentIndex = steps.indexWhere((s) => s.$1 == currentStatus);

    if (currentStatus == OrderStatus.cancelled) {
      return Container(
        padding: const EdgeInsets.all(16),
        decoration: BoxDecoration(color: Colors.red[50], borderRadius: BorderRadius.circular(12)),
        child: const Row(children: [
          Icon(Icons.cancel, color: Colors.red),
          SizedBox(width: 8),
          Text('Order Cancelled', style: TextStyle(color: Colors.red, fontWeight: FontWeight.bold)),
        ]),
      );
    }

    return Column(
      children: steps.asMap().entries.map((entry) {
        final i = entry.key;
        final step = entry.value;
        final isCompleted = i <= currentIndex;
        final isCurrent = i == currentIndex;

        return Padding(
          padding: const EdgeInsets.symmetric(vertical: 8),
          child: Row(
            children: [
              AnimatedContainer(
                duration: const Duration(milliseconds: 400),
                width: 40,
                height: 40,
                decoration: BoxDecoration(
                  color: isCompleted ? Colors.green : Colors.grey[300],
                  shape: BoxShape.circle,
                ),
                child: Icon(step.$3, color: isCompleted ? Colors.white : Colors.grey, size: 20),
              ),
              const SizedBox(width: 12),
              Expanded(
                child: Text(
                  step.$2,
                  style: TextStyle(
                    fontWeight: isCurrent ? FontWeight.bold : FontWeight.normal,
                    color: isCompleted ? Colors.black : Colors.grey,
                  ),
                ),
              ),
              if (isCurrent) const SizedBox(width: 16, height: 16, child: CircularProgressIndicator(strokeWidth: 2)),
            ],
          ),
        );
      }).toList(),
    );
  }
}
```

```dart
class _StatusChip extends StatelessWidget {

  Widget build(BuildContext context) {

        borderRadius: BorderRadius.circular(8),
```

## Step 7: Security Rules

Firestore Console → Rules:

```javascript
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /restaurants/{id} {
      allow read: if request.auth != null;
      allow write: if false;
    }
    match /restaurants/{rId}/menu/{mId} {
      allow read: if request.auth != null;
      allow write: if false;
    }
    match /orders/{orderId} {
      allow read: if request.auth != null && resource.data.userId == request.auth.uid;
      allow create: if request.auth != null && request.resource.data.userId == request.auth.uid;
      allow update: if request.auth != null && resource.data.userId == request.auth.uid;
    }
    match /users/{userId} {
      allow read, write: if request.auth != null && request.auth.uid == userId;
    }
  }
}
```
