```dart
// features/notes/domain/repositories/notes_repository.dart
abstract class NotesRepository {
  Stream<List<Note>> watchNotes(String userId);
  Future<void> add(Note note);
  Future<void> update(Note note);
  Future<void> delete(String id);
}

// features/notes/data/datasources/notes_remote_ds.dart
abstract class NotesRemoteDataSource {
  Stream<List<Map<String, dynamic>>> watchNotes(String userId);
  Future<void> addNote(Map<String, dynamic> data);
  Future<void> updateNote(String id, Map<String, dynamic> data);
  Future<void> deleteNote(String id);
}

class FirestoreNotesDataSource implements NotesRemoteDataSource {
  final FirebaseFirestore _firestore = FirebaseFirestore.instance;
  CollectionReference get _ref => _firestore.collection('notes');

  @override
  Stream<List<Map<String, dynamic>>> watchNotes(String userId) {
    return _ref
        .where('userId', isEqualTo: userId)
        .orderBy('updatedAt', descending: true)
        .snapshots()
        .map((s) => s.docs.map((d) {
              final data = d.data() as Map<String, dynamic>;
              data['id'] = d.id;
              return data;
            }).toList());
  }

  @override
  Future<void> addNote(Map<String, dynamic> data) async {
    await _ref.add(data);
  }

  @override
  Future<void> updateNote(String id, Map<String, dynamic> data) async {
    await _ref.doc(id).update(data);
  }

  @override
  Future<void> deleteNote(String id) async {
    await _ref.doc(id).delete();
  }
}

// features/notes/data/repositories/notes_repository_impl.dart
class NotesRepositoryImpl implements NotesRepository {
  final NotesRemoteDataSource remoteDS;
  NotesRepositoryImpl({required this.remoteDS});

  Note _fromMap(Map<String, dynamic> data) {
    return Note(
      id: data['id'] ?? '',
      title: data['title'] ?? '',
      content: data['content'] ?? '',
      tags: List<String>.from(data['tags'] ?? []),
      createdAt: (data['createdAt'] as Timestamp?)?.toDate() ?? DateTime.now(),
      updatedAt: (data['updatedAt'] as Timestamp?)?.toDate() ?? DateTime.now(),
      userId: data['userId'] ?? '',
    );
  }

  @override
  Stream<List<Note>> watchNotes(String userId) {
    return remoteDS.watchNotes(userId).map((list) => list.map(_fromMap).toList());
  }

  @override
  Future<void> add(Note note) {
    return remoteDS.addNote({
      'title': note.title,
      'content': note.content,
      'tags': note.tags,
      'userId': note.userId,
      'createdAt': FieldValue.serverTimestamp(),
      'updatedAt': FieldValue.serverTimestamp(),
    });
  }

  @override
  Future<void> update(Note note) {
    return remoteDS.updateNote(note.id, {
      'title': note.title,
      'content': note.content,
      'tags': note.tags,
      'updatedAt': FieldValue.serverTimestamp(),
    });
  }

  @override
  Future<void> delete(String id) => remoteDS.deleteNote(id);
}
```

```dart
// features/notes/domain/usecases/watch_notes_usecase.dart
class WatchNotesUseCase {
  final NotesRepository repository;
  WatchNotesUseCase(this.repository);

  Stream<List<Note>> call(String userId) => repository.watchNotes(userId);
}

class AddNoteUseCase {
  final NotesRepository repository;
  AddNoteUseCase(this.repository);

  Future<void> call(Note note) {
    if (note.title.trim().isEmpty) throw Exception('Title required');
    return repository.add(note);
  }
}

class DeleteNoteUseCase {
  final NotesRepository repository;
  DeleteNoteUseCase(this.repository);

  Future<void> call(String id) => repository.delete(id);
}
```

## Step 3: NotesBloc

```dart
// features/notes/presentation/bloc/notes_event.dart
sealed class NotesEvent {}
class NotesLoadRequested extends NotesEvent {
  final String userId;
  NotesLoadRequested(this.userId);
}
class NoteAddRequested extends NotesEvent {
  final Note note;
  NoteAddRequested(this.note);
}
class NoteDeleteRequested extends NotesEvent {
  final String id;
  NoteDeleteRequested(this.id);
}
class NotesUpdated extends NotesEvent {
  final List<Note> notes;
  NotesUpdated(this.notes);
}

// features/notes/presentation/bloc/notes_state.dart
sealed class NotesState extends Equatable {
  const NotesState();
  @override
  List<Object> get props => [];
}
class NotesInitial extends NotesState {}
class NotesLoading extends NotesState {}
class NotesLoaded extends NotesState {
  final List<Note> notes;
  const NotesLoaded(this.notes);
  @override
  List<Object> get props => [notes];
}
class NotesError extends NotesState {
  final String message;
  const NotesError(this.message);
  @override
  List<Object> get props => [message];
}

// features/notes/presentation/bloc/notes_bloc.dart
class NotesBloc extends Bloc<NotesEvent, NotesState> {
  final WatchNotesUseCase watchNotes;
  final AddNoteUseCase addNote;
  final DeleteNoteUseCase deleteNote;
  StreamSubscription<List<Note>>? _sub;

  NotesBloc({required this.watchNotes, required this.addNote, required this.deleteNote})
      : super(NotesInitial()) {
    on<NotesLoadRequested>(_onLoad);
    on<NotesUpdated>((event, emit) => emit(NotesLoaded(event.notes)));
    on<NoteAddRequested>((event, emit) async {
      try {
        await addNote(event.note);
      } catch (e) {
        emit(NotesError(e.toString()));
      }
    });
    on<NoteDeleteRequested>((event, emit) async {
      try {
        await deleteNote(event.id);
      } catch (e) {
        emit(NotesError(e.toString()));
      }
    });
  }

  Future<void> _onLoad(NotesLoadRequested event, Emitter emit) async {
    emit(NotesLoading());
    _sub?.cancel();
    _sub = watchNotes(event.userId).listen(
      (notes) => add(NotesUpdated(notes)),
      onError: (e) => emit(NotesError(e.toString())),
    );
  }

  @override
  Future<void> close() {
    _sub?.cancel();
    return super.close();
  }
}
```

## Step 4: get_it DI

```dart
// core/di/injection.dart
import 'package:get_it/get_it.dart';
import 'package:cloud_firestore/cloud_firestore.dart';
import '../../features/notes/data/datasources/notes_remote_ds.dart';
import '../../features/notes/data/repositories/notes_repository_impl.dart';
import '../../features/notes/domain/repositories/notes_repository.dart';
import '../../features/notes/domain/usecases/watch_notes_usecase.dart';
import '../../features/notes/domain/usecases/add_note_usecase.dart';
import '../../features/notes/domain/usecases/delete_note_usecase.dart';
import '../../features/notes/presentation/bloc/notes_bloc.dart';

final getIt = GetIt.instance;

Future<void> setupDI() async {
  // External
  getIt.registerLazySingleton(() => FirebaseFirestore.instance);

  // Data
  getIt.registerLazySingleton<NotesRemoteDataSource>(() => FirestoreNotesDataSource());

  // Domain
  getIt.registerLazySingleton<NotesRepository>(
    () => NotesRepositoryImpl(remoteDS: getIt()),
  );
  getIt.registerLazySingleton(() => WatchNotesUseCase(getIt()));
  getIt.registerLazySingleton(() => AddNoteUseCase(getIt()));
  getIt.registerLazySingleton(() => DeleteNoteUseCase(getIt()));

  // Presentation
  getIt.registerFactory(() => NotesBloc(
    watchNotes: getIt(),
    addNote: getIt(),
    deleteNote: getIt(),
  ));
}
```

```dart
// test/notes_bloc_test.dart
import 'package:bloc_test/bloc_test.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:mocktail/mocktail.dart';

class MockWatchNotes extends Mock implements WatchNotesUseCase {}
class MockAddNote extends Mock implements AddNoteUseCase {}
class MockDeleteNote extends Mock implements DeleteNoteUseCase {}

void main() {
  late MockWatchNotes mockWatch;
  late MockAddNote mockAdd;
  late MockDeleteNote mockDelete;

  setUpAll(() {
    registerFallbackValue(Note(
      id: 'fallback', title: '', content: '', tags: [],
      createdAt: DateTime.now(), updatedAt: DateTime.now(), userId: 'x',
    ));
  });

  setUp(() {
    mockWatch = MockWatchNotes();
    mockAdd = MockAddNote();
    mockDelete = MockDeleteNote();
  });

  blocTest<NotesBloc, NotesState>(
    'emits Loading then Loaded on load',
    build: () {
      when(() => mockWatch('user-1')).thenAnswer(
        (_) => Stream.value([
          Note(id: '1', title: 'Test', content: '...', tags: const [],
            createdAt: DateTime.now(), updatedAt: DateTime.now(), userId: 'user-1'),
        ]),
      );
      return NotesBloc(watchNotes: mockWatch, addNote: mockAdd, deleteNote: mockDelete);
    },
    act: (bloc) => bloc.add(NotesLoadRequested('user-1')),
    expect: () => [
      NotesLoading(),
      isA<NotesLoaded>().having((s) => s.notes.length, 'count', 1),
    ],
  );

  test('AddNoteUseCase rejects empty title', () {
    final useCase = AddNoteUseCase(MockNotesRepository());
    final note = Note(
      id: '1', title: '', content: 'body', tags: [],
      createdAt: DateTime.now(), updatedAt: DateTime.now(), userId: 'x',
    );
    expect(() => useCase(note), throwsException);
  });
}

class MockNotesRepository extends Mock implements NotesRepository {}
```

```dart
// In NotesListScreen
BlocBuilder<NotesBloc, NotesState>(
  builder: (context, state) {
    final l10n = AppLocalizations.of(context)!;

    if (state is NotesLoading) {
      return const Center(child: CircularProgressIndicator());
    }
    if (state is NotesError) {
      return Center(child: Text(state.message, style: const TextStyle(color: Colors.red)));
    }
    if (state is NotesLoaded) {
      if (state.notes.isEmpty) {
        return Center(
          child: Column(
            mainAxisAlignment: MainAxisAlignment.center,
            children: [
              const Icon(Icons.note_add, size: 100, color: Colors.grey),
              const SizedBox(height: 16),
              Text(l10n.noNotes, style: const TextStyle(fontSize: 18, color: Colors.grey)),
            ],
          ),
        );
      }

      return ListView.builder(
        itemCount: state.notes.length,
        itemBuilder: (_, i) => NoteCard(note: state.notes[i]),
      );
    }
    return const SizedBox.shrink();
  },
```
