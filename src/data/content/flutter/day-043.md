# 🌅 HOUR 1 — CONCEPT

## Flutter Desktop

Same Flutter app **Windows, macOS, Linux** pe native run hota hai.

> **Use cases:** Internal tools, dashboards, IDE-like apps, cross-platform productivity.

---

## Setup

```bash
# Enable platforms
flutter config --enable-windows-desktop
flutter config --enable-macos-desktop
flutter config --enable-linux-desktop

# Add to existing project
flutter create . --platforms=windows,macos,linux

# Run
flutter run -d windows    # ya macos / linux
```

### Build

```bash
flutter build windows --release    # → build/windows/x64/runner/Release/
flutter build macos --release      # → build/macos/Build/Products/Release/
flutter build linux --release      # → build/linux/x64/release/bundle/
```

## Window Configuration

```bash
flutter pub add window_manager
```

```dart
import 'package:window_manager/window_manager.dart';

void main() async {
  WidgetsFlutterBinding.ensureInitialized();
  await windowManager.ensureInitialized();

  WindowOptions windowOptions = const WindowOptions(
    size: Size(1200, 800),
    minimumSize: Size(800, 600),
    center: true,
    backgroundColor: Colors.transparent,
    skipTaskbar: false,
    titleBarStyle: TitleBarStyle.normal,
    title: 'My Desktop App',
  );

  windowManager.waitUntilReadyToShow(windowOptions, () async {
    await windowManager.show();
    await windowManager.focus();
  });

  runApp(const MyApp());
}
```

Methods:
```dart
windowManager.maximize();
windowManager.minimize();
windowManager.close();
windowManager.setFullScreen(true);
windowManager.setTitle('New title');
windowManager.setSize(const Size(800, 600));
```

## Keyboard Shortcuts

```dart
class MyApp extends StatelessWidget {
  const MyApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      shortcuts: {
        LogicalKeySet(LogicalKeyboardKey.control, LogicalKeyboardKey.keyN): const NewItemIntent(),
        LogicalKeySet(LogicalKeyboardKey.control, LogicalKeyboardKey.keyS): const SaveIntent(),
        LogicalKeySet(LogicalKeyboardKey.escape): const CloseDialogIntent(),
      },
      actions: {
        NewItemIntent: CallbackAction<NewItemIntent>(onInvoke: (_) {
          print('New item!');
          return null;
        }),
        SaveIntent: CallbackAction<SaveIntent>(onInvoke: (_) {
          print('Save!');
          return null;
        }),
      },
      home: const HomeScreen(),
    );
  }
}

class NewItemIntent extends Intent { const NewItemIntent(); }
class SaveIntent extends Intent { const SaveIntent(); }
class CloseDialogIntent extends Intent { const CloseDialogIntent(); }
```

> **Cross-platform:** Use `LogicalKeyboardKey.meta` for Cmd on Mac, `control` on Windows/Linux.

## File System (Desktop)

```bash
flutter pub add file_picker path_provider
```

```dart
import 'package:file_picker/file_picker.dart';

Future<void> pickFile() async {
  FilePickerResult? result = await FilePicker.platform.pickFiles(
    type: FileType.custom,
    allowedExtensions: ['pdf', 'doc'],
  );

  if (result != null) {
    final file = result.files.single;
    print(file.path);
  }
}

// Save file
Future<void> saveFile() async {
  String? outputFile = await FilePicker.platform.saveFile(
    dialogTitle: 'Save your file',
    fileName: 'output-file.pdf',
  );
}
```

## Task 2: Window Manager + Custom Title Bar (20 min)

```dart
import 'package:flutter/material.dart';
import 'package:window_manager/window_manager.dart';

void main() async {
  WidgetsFlutterBinding.ensureInitialized();
  await windowManager.ensureInitialized();

  await windowManager.waitUntilReadyToShow(
    const WindowOptions(
      size: Size(1200, 800),
      minimumSize: Size(600, 400),
      center: true,
      title: 'My Desktop App',
    ),
    () async {
      await windowManager.show();
      await windowManager.focus();
    },
  );

  runApp(const MyApp());
}

class MyApp extends StatelessWidget {
  const MyApp({super.key});
  @override
  Widget build(BuildContext context) =>
      const MaterialApp(home: WindowDemoScreen(), debugShowCheckedModeBanner: false);
}

class WindowDemoScreen extends StatelessWidget {
  const WindowDemoScreen({super.key});

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('Window Manager Demo')),
      body: Center(
        child: Column(
          mainAxisAlignment: MainAxisAlignment.center,
          children: [
            Wrap(
              spacing: 12,
              children: [
                ElevatedButton(onPressed: () => windowManager.maximize(), child: const Text('Maximize')),
                ElevatedButton(onPressed: () => windowManager.minimize(), child: const Text('Minimize')),
                ElevatedButton(
                  onPressed: () async {
                    final isFull = await windowManager.isFullScreen();
                    windowManager.setFullScreen(!isFull);
                  },
                  child: const Text('Toggle FullScreen'),
                ),
                ElevatedButton(
                  onPressed: () => windowManager.setSize(const Size(800, 600)),
                  child: const Text('Set 800×600'),
                ),
              ],
            ),
          ],
        ),
      ),
    );
  }
}
```

## Task 3: Keyboard Shortcuts + Menu Bar (20 min)

```dart
import 'package:flutter/material.dart';
import 'package:flutter/services.dart';
import 'dart:io' show exit;

class NewItemIntent extends Intent { const NewItemIntent(); }
class SaveIntent extends Intent { const SaveIntent(); }
class OpenIntent extends Intent { const OpenIntent(); }

class ShortcutsApp extends StatelessWidget {
  const ShortcutsApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      shortcuts: {
        LogicalKeySet(LogicalKeyboardKey.control, LogicalKeyboardKey.keyN): const NewItemIntent(),
        LogicalKeySet(LogicalKeyboardKey.control, LogicalKeyboardKey.keyS): const SaveIntent(),
        LogicalKeySet(LogicalKeyboardKey.control, LogicalKeyboardKey.keyO): const OpenIntent(),
      },
      home: PlatformMenuBar(
        menus: [
          PlatformMenu(label: 'File', menus: [
            PlatformMenuItem(
              label: 'New',
              shortcut: const SingleActivator(LogicalKeyboardKey.keyN, control: true),
              onSelected: () => print('Menu: New'),
            ),
            PlatformMenuItem(
              label: 'Open',
              shortcut: const SingleActivator(LogicalKeyboardKey.keyO, control: true),
              onSelected: () => print('Menu: Open'),
            ),
            PlatformMenuItem(
              label: 'Save',
              shortcut: const SingleActivator(LogicalKeyboardKey.keyS, control: true),
              onSelected: () => print('Menu: Save'),
            ),
            PlatformMenuItem(label: 'Exit', onSelected: () => exit(0)),
          ]),
          const PlatformMenu(label: 'Help', menus: []),
        ],
        child: const ShortcutsScreen(),
      ),
    );
  }
}

class ShortcutsScreen extends StatelessWidget {
  const ShortcutsScreen({super.key});

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      body: Actions(
        actions: {
          NewItemIntent: CallbackAction<NewItemIntent>(onInvoke: (_) {
            ScaffoldMessenger.of(context).showSnackBar(const SnackBar(content: Text('Ctrl+N pressed!')));
            return null;
          }),
          SaveIntent: CallbackAction<SaveIntent>(onInvoke: (_) {
            ScaffoldMessenger.of(context).showSnackBar(const SnackBar(content: Text('Ctrl+S pressed!')));
            return null;
          }),
          OpenIntent: CallbackAction<OpenIntent>(onInvoke: (_) {
            ScaffoldMessenger.of(context).showSnackBar(const SnackBar(content: Text('Ctrl+O pressed!')));
            return null;
          }),
        },
        child: Focus(
          autofocus: true,
          child: const Center(
            child: Padding(
              padding: EdgeInsets.all(40),
              child: Column(
                mainAxisAlignment: MainAxisAlignment.center,
                children: [
                  Icon(Icons.keyboard, size: 100, color: Colors.indigo),
                  SizedBox(height: 24),
                  Text('Try Shortcuts:', style: TextStyle(fontSize: 24, fontWeight: FontWeight.bold)),
                  SizedBox(height: 16),
                  Text('Ctrl+N — New'),
                  Text('Ctrl+O — Open'),
                  Text('Ctrl+S — Save'),
                  SizedBox(height: 16),
                  Text('Ya menu bar use karo top pe', style: TextStyle(color: Colors.grey)),
                ],
              ),
            ),
          ),
        ),
      ),
    );
  }
}
```

> **Test:** Press Ctrl+N, Ctrl+S — snackbars dikhne chahiye. Top menu bar bhi kaam karega.

## ❌ Common Mistakes

```dart
// MISTAKE 1: Mobile packages on desktop
// camera, geolocator → don't work or crash
// FIX: kIsWeb check + alternative

// MISTAKE 2: Window manager not initialized
windowManager.show();  // before ensureInitialized
// FIX: await windowManager.ensureInitialized() pehle

// MISTAKE 3: Hardcoded mobile sizing
SizedBox(width: 100)  // way too small on 1920px desktop
// FIX: relative sizing

// MISTAKE 4: No keyboard shortcuts on desktop
// → bad UX, users expect Ctrl+S etc
// FIX: MaterialApp.shortcuts implement
```
