# 🌅 HOUR 1 — CONCEPT

## Plugin vs Package

| | Package | Plugin |
|---|---|---|
| Dart only | ✅ | |
| Native code | ❌ | ✅ |
| Platform-specific | ❌ | ✅ |
| Examples | `intl`, `dio` | `camera`, `geolocator` |

**Plugin = package + native code (Kotlin/Swift)**.

## Modern Pattern (Federated)

`hello_plugin` me 3 layers:

1. **Public API** (`hello_plugin.dart`) — what user sees
2. **Platform Interface** — abstract contract
3. **Method Channel impl** — actual native communication

## Platform Interface

```dart
import 'package:plugin_platform_interface/plugin_platform_interface.dart';
import 'hello_plugin_method_channel.dart';

abstract class HelloPluginPlatform extends PlatformInterface {
  HelloPluginPlatform() : super(token: _token);
  static final Object _token = Object();

  static HelloPluginPlatform _instance = MethodChannelHelloPlugin();
  static HelloPluginPlatform get instance => _instance;
  static set instance(HelloPluginPlatform i) {
    PlatformInterface.verifyToken(i, _token);
    _instance = i;
  }

  Future<String?> getGreeting(String name) {
    throw UnimplementedError();
  }

  Future<int?> getRandomNumber(int max) {
    throw UnimplementedError();
  }
}
```

## Method Channel Implementation

```dart
import 'package:flutter/services.dart';
import 'hello_plugin_platform_interface.dart';

class MethodChannelHelloPlugin extends HelloPluginPlatform {
  final _channel = const MethodChannel('hello_plugin');

  @override
  Future<String?> getGreeting(String name) async {
    return await _channel.invokeMethod<String>('getGreeting', {'name': name});
  }

  @override
  Future<int?> getRandomNumber(int max) async {
    return await _channel.invokeMethod<int>('getRandomNumber', {'max': max});
  }
}
```

## Android (Kotlin)

`android/src/main/kotlin/com/yourname/hello_plugin/HelloPluginPlugin.kt`:

```kotlin
package com.yourname.hello_plugin

import io.flutter.embedding.engine.plugins.FlutterPlugin
import io.flutter.plugin.common.MethodCall
import io.flutter.plugin.common.MethodChannel
import io.flutter.plugin.common.MethodChannel.MethodCallHandler
import io.flutter.plugin.common.MethodChannel.Result
import kotlin.random.Random

class HelloPluginPlugin: FlutterPlugin, MethodCallHandler {
    private lateinit var channel: MethodChannel

    override fun onAttachedToEngine(binding: FlutterPlugin.FlutterPluginBinding) {
        channel = MethodChannel(binding.binaryMessenger, "hello_plugin")
        channel.setMethodCallHandler(this)
    }

    override fun onMethodCall(call: MethodCall, result: Result) {
        when (call.method) {
            "getGreeting" -> {
                val name = call.argument<String>("name") ?: "Unknown"
                result.success("Hello, $name from Android!")
            }
            "getRandomNumber" -> {
                val max = call.argument<Int>("max") ?: 100
                result.success(Random.nextInt(0, max))
            }
            else -> result.notImplemented()
        }
    }

    override fun onDetachedFromEngine(binding: FlutterPlugin.FlutterPluginBinding) {
        channel.setMethodCallHandler(null)
    }
}
```

## iOS (Swift)

`ios/Classes/HelloPluginPlugin.swift`:

```swift
import Flutter
import UIKit

public class HelloPluginPlugin: NSObject, FlutterPlugin {
    public static func register(with registrar: FlutterPluginRegistrar) {
        let channel = FlutterMethodChannel(name: "hello_plugin", binaryMessenger: registrar.messenger())
        let instance = HelloPluginPlugin()
        registrar.addMethodCallDelegate(instance, channel: channel)
    }

    public func handle(_ call: FlutterMethodCall, result: @escaping FlutterResult) {
        switch call.method {
        case "getGreeting":
            if let args = call.arguments as? [String: Any], let name = args["name"] as? String {
                result("Hello, \(name) from iOS!")
            } else {
                result("Hello, Unknown from iOS!")
            }
        case "getRandomNumber":
            let max = (call.arguments as? [String: Any])?["max"] as? Int ?? 100
            result(Int.random(in: 0..<max))
        default:
            result(FlutterMethodNotImplemented)
        }
    }
}
```

## Use in Example App

`example/lib/main.dart`:

```dart
import 'package:flutter/material.dart';
import 'package:hello_plugin/hello_plugin.dart';

void main() => runApp(const MyApp());

class MyApp extends StatefulWidget {
  const MyApp({super.key});
  @override
  State<MyApp> createState() => _MyAppState();
}

class _MyAppState extends State<MyApp> {
  String _greeting = '';
  int _random = 0;
  final _plugin = HelloPlugin();

  Future<void> _greet() async {
    final greeting = await _plugin.getGreeting('Flutter Dev');
    setState(() => _greeting = greeting ?? '');
  }

  Future<void> _getRandom() async {
    final n = await _plugin.getRandomNumber(max: 1000);
    setState(() => _random = n ?? 0);
  }

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      home: Scaffold(
        appBar: AppBar(title: const Text('Hello Plugin Demo')),
        body: Center(
          child: Padding(
            padding: const EdgeInsets.all(20),
            child: Column(
              mainAxisAlignment: MainAxisAlignment.center,
              children: [
                Text(_greeting, style: const TextStyle(fontSize: 22)),
                const SizedBox(height: 16),
                ElevatedButton(onPressed: _greet, child: const Text('Get Greeting')),
                const SizedBox(height: 24),
                Text('Random: $_random', style: const TextStyle(fontSize: 22)),
                const SizedBox(height: 16),
                ElevatedButton(onPressed: _getRandom, child: const Text('Get Random Number')),
              ],
            ),
          ),
        ),
      ),
    );
  }
}
```
