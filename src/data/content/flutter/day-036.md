# 🌅 HOUR 1 — CONCEPT

## Custom Painter Kya Hai?

Built-in widgets se beyond — **canvas pe direct drawing**. Charts, signature pads, custom UI shapes.

> **Socho Aise:** Built-in widgets = pre-fabricated furniture. CustomPainter = **carpenter** — kuch bhi banao with raw tools.

## Drawing Primitives

```dart
@override
void paint(Canvas canvas, Size size) {
  final paint = Paint()..color = Colors.indigo;

  // Line
  canvas.drawLine(Offset(0, 0), Offset(size.width, size.height), paint..strokeWidth = 3);

  // Circle
  canvas.drawCircle(Offset(size.width / 2, size.height / 2), 50, paint);

  // Rectangle
  canvas.drawRect(Rect.fromLTWH(10, 10, 80, 50), paint);

  // Rounded rect
  canvas.drawRRect(
    RRect.fromRectAndRadius(Rect.fromLTWH(10, 70, 80, 50), const Radius.circular(8)),
    paint,
  );

  // Arc (slice of pie)
  canvas.drawArc(
    Rect.fromCircle(center: Offset(size.width / 2, size.height / 2), radius: 80),
    0,             // start angle (radians)
    1.5,           // sweep angle
    true,          // useCenter (pie slice)
    paint,
  );

  // Path (custom shape)
  final path = Path()
    ..moveTo(0, size.height)
    ..lineTo(size.width / 2, 0)
    ..lineTo(size.width, size.height)
    ..close();
  canvas.drawPath(path, paint);
}
```

## Task 1: Pie Chart from Data (25 min)

```dart
class PieChartData {
  final String label;
  final double value;
  final Color color;
  PieChartData({required this.label, required this.value, required this.color});
}

class PieChartPainter extends CustomPainter {
  final List<PieChartData> data;

  PieChartPainter(this.data);

  @override
  void paint(Canvas canvas, Size size) {
    final total = data.fold(0.0, (sum, d) => sum + d.value);
    final center = Offset(size.width / 2, size.height / 2);
    final radius = math.min(size.width, size.height) / 2 - 20;
    final rect = Rect.fromCircle(center: center, radius: radius);

    double startAngle = -math.pi / 2;  // start at top

    for (final slice in data) {
      final sweepAngle = (slice.value / total) * 2 * math.pi;
      final paint = Paint()..color = slice.color..style = PaintingStyle.fill;

      canvas.drawArc(rect, startAngle, sweepAngle, true, paint);

      // Border
      canvas.drawArc(rect, startAngle, sweepAngle, true, Paint()..color = Colors.white..strokeWidth = 2..style = PaintingStyle.stroke);

      // Label
      final midAngle = startAngle + sweepAngle / 2;
      final labelOffset = Offset(
        center.dx + (radius * 0.6) * math.cos(midAngle),
        center.dy + (radius * 0.6) * math.sin(midAngle),
      );

      final tp = TextPainter(
        text: TextSpan(
          text: '${(slice.value / total * 100).toStringAsFixed(0)}%',
          style: const TextStyle(color: Colors.white, fontSize: 14, fontWeight: FontWeight.bold),
        ),
        textDirection: TextDirection.ltr,
      )..layout();

      tp.paint(canvas, labelOffset - Offset(tp.width / 2, tp.height / 2));

      startAngle += sweepAngle;
    }
  }

  @override
  bool shouldRepaint(PieChartPainter old) => old.data != data;
}

// Use
class PieChartScreen extends StatelessWidget {
  const PieChartScreen({super.key});

  @override
  Widget build(BuildContext context) {
    final data = [
      PieChartData(label: 'Flutter', value: 40, color: Colors.indigo),
      PieChartData(label: 'Dart', value: 25, color: Colors.green),
      PieChartData(label: 'Firebase', value: 20, color: Colors.amber),
      PieChartData(label: 'Other', value: 15, color: Colors.red),
    ];

    return Scaffold(
      appBar: AppBar(title: const Text('Pie Chart')),
      body: Center(
        child: CustomPaint(
          painter: PieChartPainter(data),
          size: const Size(300, 300),
        ),
      ),
    );
  }
}
```

## Task 2: Signature Pad (20 min)

```dart
class SignaturePad extends StatefulWidget {
  const SignaturePad({super.key});
  @override
  State<SignaturePad> createState() => _SignaturePadState();
}

class _SignaturePadState extends State<SignaturePad> {
  final List<Offset?> _points = [];

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: const Text('Signature'),
        actions: [IconButton(icon: const Icon(Icons.clear), onPressed: () => setState(() => _points.clear()))],
      ),
      body: GestureDetector(
        onPanStart: (details) => setState(() => _points.add(details.localPosition)),
        onPanUpdate: (details) => setState(() => _points.add(details.localPosition)),
        onPanEnd: (_) => setState(() => _points.add(null)),
        child: CustomPaint(
          painter: SignaturePainter(_points),
          size: Size.infinite,
        ),
      ),
    );
  }
}

class SignaturePainter extends CustomPainter {
  final List<Offset?> points;
  SignaturePainter(this.points);

  @override
  void paint(Canvas canvas, Size size) {
    final paint = Paint()
      ..color = Colors.black
      ..strokeWidth = 3
      ..strokeCap = StrokeCap.round;

    for (int i = 0; i < points.length - 1; i++) {
      if (points[i] != null && points[i + 1] != null) {
        canvas.drawLine(points[i]!, points[i + 1]!, paint);
      }
    }
  }

  @override
  bool shouldRepaint(SignaturePainter old) => old.points != points;
}
```

## Task 3: Animated Wave (15 min)

```dart
class WaveScreen extends StatefulWidget {
  const WaveScreen({super.key});
  @override
  State<WaveScreen> createState() => _WaveScreenState();
}

class _WaveScreenState extends State<WaveScreen> with SingleTickerProviderStateMixin {
  late AnimationController _controller;

  @override
  void initState() {
    super.initState();
    _controller = AnimationController(duration: const Duration(seconds: 2), vsync: this)..repeat();
  }

  @override
  void dispose() {
    _controller.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('Wave Animation')),
      body: AnimatedBuilder(
        animation: _controller,
        builder: (_, __) => CustomPaint(
          painter: WavePainter(progress: _controller.value),
          size: const Size(double.infinity, 200),
        ),
      ),
    );
  }
}

class WavePainter extends CustomPainter {
  final double progress;
  WavePainter({required this.progress});

  @override
  void paint(Canvas canvas, Size size) {
    final paint = Paint()..color = Colors.indigo..style = PaintingStyle.fill;
    final path = Path();

    path.moveTo(0, size.height);

    for (double x = 0; x <= size.width; x++) {
      final y = size.height / 2 + 30 * math.sin((x / size.width * 2 * math.pi) + progress * 2 * math.pi);
      path.lineTo(x, y);
    }

    path.lineTo(size.width, size.height);
    path.close();

    canvas.drawPath(path, paint);
  }

  @override
  bool shouldRepaint(WavePainter old) => old.progress != progress;
}
```

## Task 4: Bar Chart (15 min)

```dart
class BarChartPainter extends CustomPainter {
  final List<double> data;
  BarChartPainter(this.data);

  @override
  void paint(Canvas canvas, Size size) {
    final paint = Paint()..color = Colors.indigo;
    final barWidth = size.width / (data.length * 2);
    final maxValue = data.reduce(math.max);

    for (int i = 0; i < data.length; i++) {
      final height = (data[i] / maxValue) * size.height;
      final x = i * 2 * barWidth + barWidth / 2;
      canvas.drawRect(
        Rect.fromLTWH(x, size.height - height, barWidth, height),
        paint,
      );
    }
  }

  @override
  bool shouldRepaint(BarChartPainter old) => old.data != data;
}

// Use
CustomPaint(
  painter: BarChartPainter([20, 45, 28, 80, 99, 43, 50]),
  size: const Size(300, 200),
)
```

## 📋 Quick Revision

| API | Use |
|-----|-----|
| `CustomPainter` | Override `paint(canvas, size)` |
| `Paint()` | Color, style, strokeWidth |
| `canvas.drawLine` | Lines |

## ❌ Common Mistakes

```dart
// MISTAKE 1: Always returning true in shouldRepaint
shouldRepaint(_) => true;  // continuous repaint
// FIX: only when data changes

// MISTAKE 2: Heavy work in paint()
paint() { fetchAPI(); ... }  // ❌ called multiple times
// FIX: state se data prepare karke pass karo

// MISTAKE 3: TextPainter without layout
```
